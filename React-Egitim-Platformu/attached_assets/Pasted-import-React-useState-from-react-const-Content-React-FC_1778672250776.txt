import React, { useState } from "react";

const Content: React.FC = () => {
  const [active, setActive] = useState<"beginner" | "intermediate" | "advanced">("beginner");

  const tabClass = (tab: string) =>
    `px-4 py-2 rounded-md font-semibold transition-colors ${
      active === tab
        ? "bg-blue-600 text-white"
        : "bg-gray-200 text-gray-700 hover:bg-blue-100"
    }`;

  return (
    <section className="content p-6">
      <h2 className="text-2xl font-bold mb-4">React + TypeScript Eğitim İçeriği</h2>
      <nav className="flex gap-4 mb-6">
        <button className={tabClass("beginner")} onClick={() => setActive("beginner")}>
          Basit
        </button>
        <button className={tabClass("intermediate")} onClick={() => setActive("intermediate")}>
          Orta
        </button>
        <button className={tabClass("advanced")} onClick={() => setActive("advanced")}>
          İleri
        </button>
      </nav>

      <div className="bg-gray-50 p-4 rounded-md shadow-md">
        {active === "beginner" && (
          <div>
            <h3 className="text-lg font-semibold mb-2">Basit Örnek: Props ve State</h3>
            <pre className="bg-gray-800 text-green-200 p-3 rounded-md overflow-x-auto">
{`type GreetingProps = { name: string; age?: number };

export function Greeting({ name, age }: GreetingProps) {
  return (
    <div>
      <h2>Hello, {name}!</h2>
      {age && <p>You are {age} years old</p>}
    </div>
  );
}`}
            </pre>
          </div>
        )}

        {active === "intermediate" && (
          <div>
            <h3 className="text-lg font-semibold mb-2">Orta Örnek: API Çağrısı</h3>
            <pre className="bg-gray-800 text-yellow-200 p-3 rounded-md overflow-x-auto">
{`import React, { useState, useEffect } from "react";

type Post = { id: number; title: string };

const Posts: React.FC = () => {
  const [posts, setPosts] = useState<Post[]>([]);

  useEffect(() => {
    fetch("https://jsonplaceholder.typicode.com/posts?_limit=5")
      .then(res => res.json())
      .then(data => setPosts(data));
  }, []);

  return (
    <ul>
      {posts.map(post => (
        <li key={post.id}>{post.title}</li>
      ))}
    </ul>
  );
};`}
            </pre>
          </div>
        )}

        {active === "advanced" && (
          <div>
            <h3 className="text-lg font-semibold mb-2">İleri Örnek: Context API + Custom Hook</h3>
            <pre className="bg-gray-800 text-purple-200 p-3 rounded-md overflow-x-auto">
{`import React, { createContext, useContext, useState } from "react";

type Theme = "light" | "dark";
type ThemeContextType = { theme: Theme; toggleTheme: () => void };

const ThemeContext = createContext<ThemeContextType | undefined>(undefined);

export const ThemeProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [theme, setTheme] = useState<Theme>("light");
  const toggleTheme = () => setTheme(theme === "light" ? "dark" : "light");

  return (
    <ThemeContext.Provider value={{ theme, toggleTheme }}>
      {children}
    </ThemeContext.Provider>
  );
};`}
            </pre>
          </div>
        )}
      </div>
    </section>
  );
};

export default Content;
